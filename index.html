// -------------------------------------------------- // 🤖 Gemini APIによる単語判定 (縦・横対応版) // -------------------------------------------------- async function checkWords() { const apiKey = apiKeyInput.value.trim(); if (!apiKey) return alert("Gemini APIキーを入力してください！"); let candidates = []; // 1. 横ラインの文字列抽出（左から右） for (let r = 0; r < ROWS; r++) { let str = ""; for (let c = 0; c < COLS; c++) { if (board[r][c]) str += board[r][c]; else { if (str.length >= 2) candidates.push(str); str = ""; } } if (str.length >= 2) candidates.push(str); } // 2. 縦ラインの文字列抽出（上から下） ★追加部分★ for (let c = 0; c < COLS; c++) { let str = ""; for (let r = 0; r < ROWS; r++) { if (board[r][c]) str += board[r][c]; else { if (str.length >= 2) candidates.push(str); str = ""; } } if (str.length >= 2) candidates.push(str); } // 重複する候補を削除 candidates = [...new Set(candidates)]; if (candidates.length === 0) return alert("2文字以上並んでいる場所（縦・横）がありません！"); const prompt = `以下の文字列リストの中から日本語として意味の通じる単語を判定してください。 リスト: ${JSON.stringify(candidates)} JSON形式のみで出力: {"valid_words": [{"word": "単語", "score": 100, "comment": "一言解説", "photo_en": "English image prompt"}]}`; try { const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`; const res = await fetch(url, { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }], generationConfig: { responseMimeType: "application/json" } }) }); if (!res.ok) { if (res.status === 400 || res.status === 403) throw new Error("APIキーが無効または間違っています。"); if (res.status === 429) throw new Error("利用制限（リクエスト過多）です。少し待ってからお試しください。"); throw new Error(`通信エラーが発生しました (コード: ${res.status})`); } const data = await res.json(); const rawText = data.candidates?.[0]?.content?.parts?.[0]?.text; if (!rawText) throw new Error("Geminiからの回答を受け取れませんでした。"); const result = JSON.parse(rawText); if (result.valid_words && result.valid_words.length > 0) { playSound('success'); result.valid_words.forEach(w => { addLog(w.word, w.score, w.comment); updateScore(w.score); if (w.photo_en) showImage(w.photo_en); }); } else { playSound('fail'); addLog("該当なし", 0, "意味の通じる単語が見つかりませんでした..."); } } catch(e) { playSound('fail'); alert("【判定エラー】\n" + e.message); } }
